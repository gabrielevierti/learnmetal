In pratica

Un gioco 2D#

Finora è stato tutto tecnica.
Questo capitolo mette al lavoro una manciata di quelle tecniche in qualcosa di giocabile: un piccolo gioco in cui si rompono mattoni.
Una racchetta, una palla, un muro di mattoni.
È minuscolo di proposito, così l'intera struttura sta in un capitolo e puoi vedere come si incontrano un renderer e la logica di gioco.

Che cosa ci serve#

  • Uno sprite renderer che disegna molti rettangoli colorati e con texture in una sola draw call.
  • Una camera ortografica in unità simili ai pixel.
  • Un game loop con delta time e input.
  • Il rilevamento delle collisioni tra un cerchio e dei rettangoli.

Lo sprite renderer#

Uno sprite è un rettangolo con una posizione, una dimensione, un colore e una regione di texture.
Usiamo l'instancing: un rettangolo unitario generato nella vertex function e un'istanza per sprite.

struct SpriteInstance {
    float2 position;     // top-left corner, in game units
    float2 size;
    float4 color;
    float4 uvRect;       // xy = origin, zw = size, within the texture
};

struct SpriteOut {
    float4 position [[position]];
    float2 uv;
    float4 color;
};

vertex SpriteOut vertex_sprite(uint vertexID [[vertex_id]],
                               uint instanceID [[instance_id]],
                               const device SpriteInstance *sprites [[buffer(0)]],
                               constant float4x4 &projection [[buffer(1)]]) {
    const float2 corners[4] = { {0, 0}, {1, 0}, {0, 1}, {1, 1} };
    SpriteInstance sprite = sprites[instanceID];
    float2 corner = corners[vertexID];

    SpriteOut out;
    out.position = projection * float4(sprite.position + corner * sprite.size, 0.0, 1.0);
    out.uv = sprite.uvRect.xy + corner * sprite.uvRect.zw;
    out.color = sprite.color;
    return out;
}

fragment float4 fragment_sprite(SpriteOut in [[stage_in]],
                                texture2d<float> atlas [[texture(0)]],
                                sampler s [[sampler(0)]]) {
    return atlas.sample(s, in.uv) * in.color;
}

Dal lato Swift il renderer raccoglie gli sprite durante il frame e li disegna tutti insieme:

struct SpriteInstance {
    var position: SIMD2<Float>
    var size: SIMD2<Float>
    var color: SIMD4<Float>
    var uvRect: SIMD4<Float> = [0, 0, 1, 1]
}

final class SpriteBatch {
    private var sprites: [SpriteInstance] = []
    private let buffer: MTLBuffer
    private let capacity = 4096

    init(device: MTLDevice) {
        buffer = device.makeBuffer(length: MemoryLayout<SpriteInstance>.stride * capacity,
                                   options: .storageModeShared)!
    }

    func add(_ sprite: SpriteInstance) {
        if sprites.count < capacity { sprites.append(sprite) }
    }

    func flush(encoder: MTLRenderCommandEncoder) {
        guard !sprites.isEmpty else { return }
        buffer.contents().copyMemory(from: sprites,
                                     byteCount: MemoryLayout<SpriteInstance>.stride * sprites.count)
        encoder.setVertexBuffer(buffer, offset: 0, index: 0)
        encoder.drawPrimitives(type: .triangleStrip, vertexStart: 0, vertexCount: 4,
                               instanceCount: sprites.count)
        sprites.removeAll(keepingCapacity: true)
    }
}

struct SpriteInstance {
    simd::float2 position;
    simd::float2 size;
    simd::float4 color;
    simd::float4 uvRect = {0, 0, 1, 1};
};

class SpriteBatch {
public:
    explicit SpriteBatch(MTL::Device* device) {
        buffer = device->newBuffer(sizeof(SpriteInstance) * capacity, MTL::ResourceStorageModeShared);
        sprites.reserve(capacity);
    }

    void add(const SpriteInstance& sprite) {
        if (sprites.size() < capacity) sprites.push_back(sprite);
    }

    void flush(MTL::RenderCommandEncoder* encoder) {
        if (sprites.empty()) return;
        std::memcpy(buffer->contents(), sprites.data(), sizeof(SpriteInstance) * sprites.size());
        encoder->setVertexBuffer(buffer, 0, 0);
        encoder->drawPrimitives(MTL::PrimitiveTypeTriangleStrip, NS::UInteger(0), NS::UInteger(4),
                                NS::UInteger(sprites.size()));
        sprites.clear();
    }

private:
    static constexpr size_t capacity = 4096;
    std::vector<SpriteInstance> sprites;
    MTL::Buffer* buffer;
};

Usa una pipeline con alpha blending.
Metti tutte le immagini del gioco in un'unica texture, un atlas, e scegli le regioni con uvRect, così l'intero gioco è una sola draw call.
Per i rettangoli a tinta unita collega una texture bianca da 1 × 1.

Il campo di gioco#

Dai al gioco una dimensione logica fissa, indipendente dalla finestra, con l'origine in alto a sinistra e y verso il basso, cosa comoda in 2D:

let fieldSize = SIMD2<Float>(800, 600)
var projection = float4x4(orthographicLeft: 0, right: fieldSize.x,
                          bottom: fieldSize.y, top: 0,
                          near: -1, far: 1)

const simd::float2 fieldSize = {800, 600};
simd::float4x4 projection = orthographic(0, fieldSize.x,      // left, right
                                         fieldSize.y, 0,      // bottom, top: swapped to flip y
                                         -1, 1);

Scambiare bottom e top ribalta l'asse y.

Gli oggetti di gioco#

struct Brick {
    var position: SIMD2<Float>
    var size: SIMD2<Float>
    var color: SIMD4<Float>
    var isAlive = true
}

struct Ball {
    var position: SIMD2<Float>       // center
    var velocity: SIMD2<Float>
    var radius: Float = 10
    var isStuck = true                // resting on the paddle until launched
}

final class Game {
    var paddlePosition = SIMD2<Float>(350, 570)
    let paddleSize = SIMD2<Float>(100, 16)
    let paddleSpeed: Float = 500
    var ball = Ball(position: [400, 560], velocity: [120, -350])
    var bricks: [Brick] = []
    var lives = 3

    func loadLevel(rows: Int, columns: Int) {
        let size = SIMD2<Float>(800 / Float(columns), 24)
        let palette: [SIMD4<Float>] = [[0.9, 0.3, 0.3, 1], [0.9, 0.6, 0.2, 1],
                                       [0.3, 0.8, 0.4, 1], [0.3, 0.5, 0.9, 1]]
        bricks = (0..<rows).flatMap { row in
            (0..<columns).map { column in
                Brick(position: [Float(column) * size.x, 60 + Float(row) * size.y],
                      size: size - 2,
                      color: palette[row % palette.count])
            }
        }
    }
}

struct Brick {
    simd::float2 position;
    simd::float2 size;
    simd::float4 color;
    bool isAlive = true;
};

struct Ball {
    simd::float2 position;        // center
    simd::float2 velocity;
    float radius = 10;
    bool isStuck = true;          // resting on the paddle until launched
};

class Game {
public:
    simd::float2 paddlePosition = {350, 570};
    const simd::float2 paddleSize = {100, 16};
    const float paddleSpeed = 500;
    Ball ball{{400, 560}, {120, -350}};
    std::vector<Brick> bricks;
    int lives = 3;

    void loadLevel(int rows, int columns) {
        const simd::float2 size = {800.0f / columns, 24};
        const simd::float4 palette[4] = {{0.9f, 0.3f, 0.3f, 1}, {0.9f, 0.6f, 0.2f, 1},
                                         {0.3f, 0.8f, 0.4f, 1}, {0.3f, 0.5f, 0.9f, 1}};
        bricks.clear();
        for (int row = 0; row < rows; row++) {
            for (int column = 0; column < columns; column++) {
                bricks.push_back({{column * size.x, 60 + row * size.y}, size - 2.0f, palette[row % 4]});
            }
        }
    }

    void update(float deltaTime, float input, bool launch);
};

Un livello è soltanto dati.
Caricare i livelli da un file di testo in cui ogni cifra è un tipo di mattone è un'estensione facile.

Update#

extension Game {
    func update(deltaTime: Float, input: Float, launch: Bool) {
        // Paddle: input is -1, 0 or 1.
        paddlePosition.x += input * paddleSpeed * deltaTime
        paddlePosition.x = min(max(paddlePosition.x, 0), 800 - paddleSize.x)

        if ball.isStuck {
            ball.position = [paddlePosition.x + paddleSize.x / 2, paddlePosition.y - ball.radius]
            if launch { ball.isStuck = false }
            return
        }

        ball.position += ball.velocity * deltaTime

        // Walls
        if ball.position.x < ball.radius || ball.position.x > 800 - ball.radius {
            ball.velocity.x = -ball.velocity.x
            ball.position.x = min(max(ball.position.x, ball.radius), 800 - ball.radius)
        }
        if ball.position.y < ball.radius {
            ball.velocity.y = abs(ball.velocity.y)
        }

        // Bricks
        for i in bricks.indices where bricks[i].isAlive {
            if let normal = collide(ball, rectPosition: bricks[i].position, rectSize: bricks[i].size) {
                bricks[i].isAlive = false
                ball.velocity = reflect(ball.velocity, n: normal)
                break
            }
        }

        // Paddle: the bounce angle depends on where the ball hits.
        if ball.velocity.y > 0,
           collide(ball, rectPosition: paddlePosition, rectSize: paddleSize) != nil {
            let center = paddlePosition.x + paddleSize.x / 2
            let offset = (ball.position.x - center) / (paddleSize.x / 2)      // -1...1
            let speed = length(ball.velocity)
            ball.velocity = normalize(SIMD2<Float>(offset, -1.2)) * speed
        }

        // Lost the ball
        if ball.position.y > 600 + ball.radius {
            lives -= 1
            ball.isStuck = true
            ball.velocity = [120, -350]
        }
    }
}

void Game::update(float deltaTime, float input, bool launch) {
    // Paddle: input is -1, 0 or 1.
    paddlePosition.x += input * paddleSpeed * deltaTime;
    paddlePosition.x = std::clamp(paddlePosition.x, 0.0f, 800 - paddleSize.x);

    if (ball.isStuck) {
        ball.position = {paddlePosition.x + paddleSize.x / 2, paddlePosition.y - ball.radius};
        if (launch) ball.isStuck = false;
        return;
    }

    ball.position += ball.velocity * deltaTime;

    // Walls
    if (ball.position.x < ball.radius || ball.position.x > 800 - ball.radius) {
        ball.velocity.x = -ball.velocity.x;
        ball.position.x = std::clamp(ball.position.x, ball.radius, 800 - ball.radius);
    }
    if (ball.position.y < ball.radius) {
        ball.velocity.y = std::abs(ball.velocity.y);
    }

    // Bricks
    for (Brick& brick : bricks) {
        if (!brick.isAlive) continue;
        simd::float2 normal;
        if (collide(ball, brick.position, brick.size, normal)) {
            brick.isAlive = false;
            ball.velocity = simd::reflect(ball.velocity, normal);
            break;
        }
    }

    // Paddle: the bounce angle depends on where the ball hits.
    simd::float2 unused;
    if (ball.velocity.y > 0 && collide(ball, paddlePosition, paddleSize, unused)) {
        float center = paddlePosition.x + paddleSize.x / 2;
        float offset = (ball.position.x - center) / (paddleSize.x / 2);      // -1...1
        float speed = simd::length(ball.velocity);
        ball.velocity = simd::normalize(simd::float2{offset, -1.2f}) * speed;
    }

    // Lost the ball
    if (ball.position.y > 600 + ball.radius) {
        lives -= 1;
        ball.isStuck = true;
        ball.velocity = {120, -350};
    }
}

Dirigere la palla in base a dove tocca la racchetta è ciò che rende il gioco controllabile.
Una riflessione puramente fisica risulterebbe senza vita.

Cerchio contro rettangolo#

Trova il punto del rettangolo più vicino al centro del cerchio con un clamp.
Se quel punto dista meno di un raggio, i due si sovrappongono.
La direzione dal punto più vicino al centro ci dice quale lato è stato colpito:

func collide(_ ball: Ball, rectPosition: SIMD2<Float>, rectSize: SIMD2<Float>) -> SIMD2<Float>? {
    let closest = simd_clamp(ball.position, rectPosition, rectPosition + rectSize)
    let difference = ball.position - closest
    guard length_squared(difference) < ball.radius * ball.radius else { return nil }

    // Pick the dominant axis as the collision normal.
    if abs(difference.x) > abs(difference.y) {
        return [difference.x > 0 ? 1 : -1, 0]
    } else {
        return [0, difference.y > 0 ? 1 : -1]
    }
}

// Returns true on overlap and writes the collision normal.
bool collide(const Ball& ball, simd::float2 rectPosition, simd::float2 rectSize, simd::float2& normal) {
    simd::float2 closest = simd::clamp(ball.position, rectPosition, rectPosition + rectSize);
    simd::float2 difference = ball.position - closest;
    if (simd::length_squared(difference) >= ball.radius * ball.radius) return false;

    // Pick the dominant axis as the collision normal.
    if (std::abs(difference.x) > std::abs(difference.y)) {
        normal = {difference.x > 0 ? 1.0f : -1.0f, 0};
    } else {
        normal = {0, difference.y > 0 ? 1.0f : -1.0f};
    }
    return true;
}

Ad alta velocità una palla può attraversare un mattone sottile tra un frame e l'altro.
Eseguire l'aggiornamento in più piccoli sotto-passi per frame è la protezione più semplice.

Render#

func render(game: Game, batch: SpriteBatch, encoder: MTLRenderCommandEncoder) {
    for brick in game.bricks where brick.isAlive {
        batch.add(SpriteInstance(position: brick.position, size: brick.size, color: brick.color))
    }
    batch.add(SpriteInstance(position: game.paddlePosition, size: game.paddleSize,
                             color: [0.9, 0.9, 0.9, 1]))
    let r = game.ball.radius
    batch.add(SpriteInstance(position: game.ball.position - r, size: [2 * r, 2 * r],
                             color: [1, 1, 1, 1], uvRect: ballUVRect))

    encoder.setRenderPipelineState(spritePipelineState)
    encoder.setVertexBytes(&projection, length: MemoryLayout<float4x4>.stride, index: 1)
    encoder.setFragmentTexture(atlasTexture, index: 0)
    encoder.setFragmentSamplerState(samplerState, index: 0)
    batch.flush(encoder: encoder)
}

void render(const Game& game, SpriteBatch& batch, MTL::RenderCommandEncoder* encoder) {
    for (const Brick& brick : game.bricks) {
        if (brick.isAlive) batch.add({brick.position, brick.size, brick.color});
    }
    batch.add({game.paddlePosition, game.paddleSize, {0.9f, 0.9f, 0.9f, 1}});
    float r = game.ball.radius;
    batch.add({game.ball.position - r, {2 * r, 2 * r}, {1, 1, 1, 1}, ballUVRect});

    encoder->setRenderPipelineState(spritePipelineState);
    encoder->setVertexBytes(&projection, sizeof(simd::float4x4), 1);
    encoder->setFragmentTexture(atlasTexture, 0);
    encoder->setFragmentSamplerState(samplerState, 0);
    batch.flush(encoder);
}

Renderlo piacevole#

Il gioco funziona.
Ciò che lo rende divertente è la rifinitura, e ogni voce qui sotto riusa una tecnica di questa serie:

  • Particelle: una scia con blending additivo dietro la palla, disegnata con lo stesso gruppo di sprite.
  • Scosse dello schermo e lampi: disegna il gioco in una texture e deformala in un pass di post-processing per qualche frame dopo un colpo.
  • Bagliore: il bloom sui mattoni luminosi.
  • Punteggio e vite: il testo con atlas di Rendering del testo.
  • Power-up: sprite che cadono dai mattoni distrutti e, se presi, cambiano la dimensione della racchetta o la velocità della palla.
  • Suono: AVAudioEngine o AVAudioPlayer per gli effetti dei colpi.
